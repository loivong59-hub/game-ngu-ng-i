<script>

    /* =========================
       BIẾN GAME
    ========================= */

    const game =
        document.getElementById("game");

    const player =
        document.getElementById("player");

    const scoreText =
        document.getElementById("score");

    const highScoreText =
        document.getElementById("highScore");

    const gameOverScreen =
        document.getElementById("gameOver");

    const finalScore =
        document.getElementById("finalScore");


    let playerX = 172;

    let score = 0;

    let highScore =
        localStorage.getItem("highScore") || 0;

    let gameRunning = true;

    let speed = 5;

    let obstacles = [];

    let coins = [];

    let keys = {};


    highScoreText.innerText =
        "Kỷ lục: " + highScore;


    /* =========================
       ĐIỀU KHIỂN
    ========================= */

    document.addEventListener(
        "keydown",
        function(e) {

            keys[e.key] = true;

        }
    );


    document.addEventListener(
        "keyup",
        function(e) {

            keys[e.key] = false;

        }
    );


    /* =========================
       TẠO VẬT CẢN
    ========================= */

    function createObstacle() {

        if (!gameRunning)
            return;


        const obstacle =
            document.createElement("div");

        obstacle.className =
            "obstacle";


        let x =
            Math.random() * 330;


        obstacle.style.left =
            x + "px";


        obstacle.style.top =
            "-80px";


        game.appendChild(obstacle);


        obstacles.push({

            element: obstacle,

            x: x,

            y: -80

        });

    }


    /* =========================
       TẠO COIN
    ========================= */

    function createCoin() {

        if (!gameRunning)
            return;


        const coin =
            document.createElement("div");

        coin.className =
            "coin";


        let x =
            Math.random() * 360;


        coin.style.left =
            x + "px";


        coin.style.top =
            "-40px";


        game.appendChild(coin);


        coins.push({

            element: coin,

            x: x,

            y: -40

        });

    }


    /* =========================
       KIỂM TRA VA CHẠM
    ========================= */

    function collision(a, b) {

        return !(
            a.x + a.width < b.x ||
            a.x > b.x + b.width ||
            a.y + a.height < b.y ||
            a.y > b.y + b.height
        );

    }


    /* =========================
       GAME LOOP
    ========================= */

    function gameLoop() {

        if (!gameRunning)
            return;


        /* DI CHUYỂN PLAYER */

        if (keys["ArrowLeft"]) {

            playerX -= 7;

        }


        if (keys["ArrowRight"]) {

            playerX += 7;

        }


        /* GIỚI HẠN */

        if (playerX < 0)
            playerX = 0;


        if (playerX > 345)
            playerX = 345;


        player.style.left =
            playerX + "px";


        /* PLAYER RECT */

        const playerRect = {

            x: playerX,

            y: game.clientHeight - 105,

            width: 55,

            height: 75

        };


        /* =================
           OBSTACLE
        ================= */

        obstacles.forEach(
            (obstacle, index) => {

                obstacle.y += speed;


                obstacle.element.style.top =
                    obstacle.y + "px";


                const obstacleRect = {

                    x: obstacle.x,

                    y: obstacle.y,

                    width: 55,

                    height: 65

                };


                if (
                    collision(
                        playerRect,
                        obstacleRect
                    )
                ) {

                    endGame();

                }


                if (obstacle.y > 700) {

                    obstacle.element.remove();

                    obstacles.splice(
                        index,
                        1
                    );

                    score++;

                    scoreText.innerText =
                        "Điểm: " + score;


                    /* TĂNG TỐC */

                    if (
                        score % 10 === 0
                    ) {

                        speed += 0.5;

                    }

                }

            }
        );


        /* =================
           COINS
        ================= */

        coins.forEach(
            (coin, index) => {

                coin.y += speed;


                coin.element.style.top =
                    coin.y + "px";


                const coinRect = {

                    x: coin.x,

                    y: coin.y,

                    width: 30,

                    height: 30

                };


                if (
                    collision(
                        playerRect,
                        coinRect
                    )
                ) {

                    score += 5;


                    scoreText.innerText =
                        "Điểm: " + score;


                    coin.element.remove();

                    coins.splice(
                        index,
                        1
                    );

                }


                if (coin.y > 700) {

                    coin.element.remove();

                    coins.splice(
                        index,
                        1
                    );

                }

            }
        );


        requestAnimationFrame(
            gameLoop
        );

    }


    /* =========================
       GAME OVER
    ========================= */

    function endGame() {

        gameRunning = false;


        finalScore.innerText =
            score;


        gameOverScreen.style.display =
            "flex";


        if (
            score > highScore
        ) {

            highScore = score;


            localStorage.setItem(
                "highScore",
                highScore
            );

        }


        highScoreText.innerText =
            "Kỷ lục: " + highScore;

    }


    /* =========================
       RESTART
    ========================= */

    function restartGame() {

        obstacles.forEach(
            obstacle =>
                obstacle.element.remove()
        );


        coins.forEach(
            coin =>
                coin.element.remove()
        );


        obstacles = [];

        coins = [];


        score = 0;

        speed = 5;

        playerX = 172;

        gameRunning = true;


        scoreText.innerText =
            "Điểm: 0";


        player.style.left =
            playerX + "px";


        gameOverScreen.style.display =
            "none";


        gameLoop();

    }


    /* =========================
       TẠO VẬT CẢN ĐỊNH KỲ
    ========================= */

    setInterval(
        createObstacle,
        1200
    );


    /* =========================
       TẠO COIN
    ========================= */

    setInterval(
        createCoin,
        1800
    );


    /* =========================
       START GAME
    ========================= */

    gameLoop();

</script>
